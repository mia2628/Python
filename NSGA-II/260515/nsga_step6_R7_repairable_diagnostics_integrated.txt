def minmax(x):
    x = np.asarray(x, dtype=float)
    lo, hi = np.nanmin(x), np.nanmax(x)
    if hi - lo < 1e-12:
        return np.zeros_like(x, dtype=float)
    return (x - lo) / (hi - lo)

def safe_ratio(num, den):
    num = float(num)
    den = float(den)
    if abs(den) < 1e-12:
        return 0.0
    return num / den

def print_array_summary(name, arr):
    arr = np.asarray(arr, dtype=float)
    finite = arr[np.isfinite(arr)]
    if finite.size == 0:
        print(f"[DIAG] {name}: no finite values")
        return
    print(
        f"[DIAG] {name}: "
        f"min={np.min(finite):.6g} | "
        f"p50={np.percentile(finite, 50):.6g} | "
        f"p90={np.percentile(finite, 90):.6g} | "
        f"max={np.max(finite):.6g} | "
        f"mean={np.mean(finite):.6g}"
    )

def convert_failure_rate_to_annual_lambda(fr_raw, unit, annual_operating_hours):
    fr_raw = np.asarray(fr_raw, dtype=float)
    fr_raw = np.nan_to_num(fr_raw, nan=0.0, posinf=0.0, neginf=0.0)
    fr_raw = np.clip(fr_raw, 0.0, None)

    unit = str(unit).strip().lower()
    annual_operating_hours = float(annual_operating_hours)
    if not np.isfinite(annual_operating_hours) or annual_operating_hours <= 0:
        raise ValueError("ANNUAL_OPERATING_HOURS must be a positive finite number.")

    if unit == "per_million_hours":
        return fr_raw * annual_operating_hours / 1_000_000.0

    if unit == "annual_rate":
        return fr_raw.copy()

    if unit == "annual_probability":
        p = np.clip(fr_raw, 0.0, 0.999999999)
        return -np.log(1.0 - p)

    raise ValueError(
        "Unknown FAILURE_RATE_UNIT. Use one of: "
        "'per_million_hours', 'annual_rate', 'annual_probability'."
    )

def build_qpa_array(df, col_qpa, n_rows, default_qpa=1.0):
    default_qpa = float(default_qpa)
    if not np.isfinite(default_qpa) or default_qpa <= 0:
        raise ValueError("DEFAULT_QPA must be a positive finite number.")

    if col_qpa is None:
        qpa = np.full(n_rows, default_qpa, dtype=float)
        print(f"[QPA] No QPA column found. Using DEFAULT_QPA={default_qpa:.3g} for all parts.")
        return qpa

    qpa = pd.to_numeric(df[col_qpa], errors="coerce").fillna(default_qpa).to_numpy(float)
    qpa = np.nan_to_num(qpa, nan=default_qpa, posinf=default_qpa, neginf=default_qpa)
    qpa = np.where(qpa > 0, qpa, default_qpa).astype(float)
    print(f"[QPA] Using column '{col_qpa}' as quantity per equipment/system.")
    return qpa

    default_value = float(default_value)
    if not np.isfinite(default_value) or default_value < 0:

    if col_name is None:

        print(f"[{label}] No column found. Using default={default_value:.6g} for all parts.")

    arr = np.nan_to_num(arr, nan=default_value, posinf=default_value, neginf=default_value)

    print(f"[{label}] Using column '{col_name}'.")

def build_repairable_flag_array(df, col_name, n_rows, default_value=0):
    if col_name is None:
        arr = np.full(n_rows, int(default_value), dtype=int)
        print(f"[Repairable_Flag] No column found. Using default={int(default_value)} for all parts.")
        return arr

    true_values = {"1", "true", "t", "yes", "y", "repairable", "수리가능", "수리", "가능"}
    false_values = {"0", "false", "f", "no", "n", "nonrepairable", "not repairable", "폐기", "불가", "소모"}

    raw = df[col_name].fillna(default_value).astype(str).str.strip().str.lower()
    arr = np.full(n_rows, int(default_value), dtype=int)

    for i, v in enumerate(raw.to_numpy()):
        if v in true_values:
            arr[i] = 1
        elif v in false_values:
            arr[i] = 0
        else:
            num = pd.to_numeric(v, errors="coerce")
            arr[i] = 1 if pd.notna(num) and float(num) > 0 else int(default_value)

    print(f"[Repairable_Flag] Using column '{col_name}'.")
    return arr


def apply_sibling_normalization(lambda_base, parent_id_all, part_id_all, spare_idx):
    lambda_base = np.asarray(lambda_base, dtype=float)
    lambda_eff = lambda_base.copy()

    parent_group_map = {}
    for local_idx, global_idx in enumerate(spare_idx):
        p = parent_id_all[global_idx]
        p = str(p) if pd.notna(p) else f"ROOT_{part_id_all[global_idx]}"
        parent_group_map.setdefault(p, []).append(local_idx)

    for _, members in parent_group_map.items():
        if len(members) <= 1:
            continue
        sub = lambda_base[members]
        total = sub.sum()
        if total > 1e-12:
            shares = sub / total
            lambda_eff[members] = total * shares / len(members) * 1.2
        else:
            lambda_eff[members] = lambda_base[members]

    return lambda_eff

    if any(k in s for k in ["intermediate", "support", "shop", "base", "i-level", "i level", "direct support"]):

def pm_cycle_to_days(v):
    if pd.isna(v):
        return np.nan

    if isinstance(v, (int, float, np.integer, np.floating)):
        x = float(v)
        if x <= 0:
            return np.nan
        return x if x <= 400 else np.nan

    s = str(v).strip().lower()
    if s in ["", "nan", "none", "null"]:
        return np.nan

    if "분기" in s or "quarter" in s:
        return 365.0 / 4.0
    if "반기" in s or "semi" in s or "half" in s:
        return 365.0 / 2.0
    if "연" in s or "annual" in s or "year" in s:
        return 365.0

    m = re.search(r"(\d+(\.\d+)?)\s*(day|days|d|month|months|m|year|years|y)", s)
    if m:
        val = float(m.group(1))
        unit = m.group(3)
        if unit in ["day", "days", "d"]:
            return val
        if unit in ["month", "months", "m"]:
            return val * 30.4167
        if unit in ["year", "years", "y"]:
            return val * 365.0

    m2 = re.search(r"(\d+(\.\d+)?)", s)
    if m2:
        return float(m2.group(1))

    return np.nan

def poisson_tail_prob(mu, s):
    mu = float(max(mu, 0.0))
    s = int(max(s, 0))
    if mu < 1e-12:
        return 0.0

    term = math.exp(-mu)
    cdf = term
    for k in range(1, s + 1):
        term *= mu / k
        cdf += term
    tail = 1.0 - cdf
    return min(max(tail, 0.0), 1.0)

def stockout_prob_from_lead_demand(mu_lead, stock_units):
    stock_units = int(max(stock_units, 0))
    if stock_units <= 0:
        return 1.0
    return poisson_tail_prob(mu_lead, stock_units)

def get_shortage_exposure_hours(lead_hours):
    lead_hours = np.asarray(lead_hours, dtype=float)
    lead_hours = np.nan_to_num(lead_hours, nan=0.0, posinf=0.0, neginf=0.0)
    floor_h = float(LEAD_TIME_SHORTAGE_FLOOR_H)
    if not np.isfinite(floor_h) or floor_h <= 0:
        raise ValueError("LEAD_TIME_SHORTAGE_FLOOR_H must be a positive finite number.")
    return np.maximum(lead_hours, floor_h)


def apply_repairable_time_split(
    restore_base_h,
    lead_time_h,
    repairable_flag,
    replace_time_h,
    repair_tat_h,
):
    restore_base_h = np.asarray(restore_base_h, dtype=float)
    lead_time_h = np.asarray(lead_time_h, dtype=float)
    repairable_flag = np.asarray(repairable_flag, dtype=int)
    replace_time_h = np.asarray(replace_time_h, dtype=float)
    repair_tat_h = np.asarray(repair_tat_h, dtype=float)

    restore_base_h = np.nan_to_num(restore_base_h, nan=0.0, posinf=0.0, neginf=0.0)
    lead_time_h = np.nan_to_num(lead_time_h, nan=0.0, posinf=0.0, neginf=0.0)
    replace_time_h = np.nan_to_num(replace_time_h, nan=0.0, posinf=0.0, neginf=0.0)
    repair_tat_h = np.nan_to_num(repair_tat_h, nan=0.0, posinf=0.0, neginf=0.0)

    repairable_mask = repairable_flag > 0
    replace_valid = replace_time_h > 0
    tat_valid = repair_tat_h > 0

    if APPLY_REPAIRABLE_TIME_SPLIT:
        restore_used_h = np.where(
            repairable_mask & replace_valid,
            replace_time_h,
            restore_base_h,
        )
        shortage_raw_h = np.where(
            repairable_mask & tat_valid,
            repair_tat_h,
            lead_time_h,
        )
    else:
        restore_used_h = restore_base_h.copy()
        shortage_raw_h = lead_time_h.copy()

    restore_used_h = np.clip(restore_used_h, 0.0, None)
    shortage_exposure_h = get_shortage_exposure_hours(shortage_raw_h)

    restore_split_applied = (repairable_mask & replace_valid & APPLY_REPAIRABLE_TIME_SPLIT).astype(int)
    exposure_split_applied = (repairable_mask & tat_valid & APPLY_REPAIRABLE_TIME_SPLIT).astype(int)

    return restore_used_h, shortage_exposure_h, restore_split_applied, exposure_split_applied

def long_dynamic_cap_from_lambda(lam_2y):
    base = int(np.ceil(lam_2y))
    if lam_2y < 1.0:
        buffer_i = 1
        abs_cap_i = ABS_CAP_SMALL
    elif lam_2y < 2.5:
        buffer_i = 1
        abs_cap_i = ABS_CAP_MED
    else:
        buffer_i = 2
        abs_cap_i = ABS_CAP_LARGE
    total_cap_i = min(base + buffer_i, abs_cap_i)
    return max(total_cap_i, 1)


def stock_cap_from_pipeline_mu(mu_pipe, lam_2y, is_long=False, repairable=False):
    """
    Return an NSGA-II stock search upper bound.

    - mu_pipe: expected demand during the relevant shortage exposure window.
      For repairable items this is usually Repair_TAT_H demand.
    - lam_2y: expected demand over the whole analysis period.
    - is_long: long lead/TAT item flag.
    - repairable: whether the item is repairable and may create repair pipeline stock demand.

    The function preserves the previous cap philosophy but makes repairable
    pipeline demand visible even for long-lead items.
    """
    mu_pipe = float(max(mu_pipe, 0.0))
    lam_2y = float(max(lam_2y, 0.0))

    if lam_2y < 1.0:
        abs_cap_i = ABS_CAP_SMALL
    elif lam_2y < 2.5:
        abs_cap_i = ABS_CAP_MED
    else:
        abs_cap_i = ABS_CAP_LARGE

    z = float(STOCK_CAP_SAFETY_Z)
    buffer_units = float(STOCK_CAP_BUFFER_UNITS)
    if not np.isfinite(z) or z < 0:
        raise ValueError("STOCK_CAP_SAFETY_Z must be a non-negative finite number.")
    if not np.isfinite(buffer_units) or buffer_units < 0:
        raise ValueError("STOCK_CAP_BUFFER_UNITS must be a non-negative finite number.")

    exposure_cap = int(np.ceil(mu_pipe + z * np.sqrt(max(mu_pipe, 1e-9)) + buffer_units))
    exposure_cap = max(exposure_cap, 1)

    if is_long:
        # Previous long-lead behavior is retained as a lower bound.
        legacy_long_cap = long_dynamic_cap_from_lambda(lam_2y)
        cap = max(legacy_long_cap, exposure_cap)
    else:
        cap = exposure_cap

    # Repairable items should be sized from repair-pipeline demand, but the
    # search space still needs a hard cap for stability.
    if repairable and APPLY_REPAIRABLE_STOCK_CAP:
        cap = max(cap, exposure_cap)

    return int(min(max(cap, 1), abs_cap_i))

def get_ancestor_chain(idx):
    chain_rev = [idx]
    seen = {idx}
    cur_parent = parent_id_all[idx]
    hop = 0
    while pd.notna(cur_parent) and str(cur_parent) not in ["", "nan", "None", "null"]:
        cur_parent = str(cur_parent)
        if cur_parent not in part_to_idx:
            break
        pidx = part_to_idx[cur_parent]
        if pidx in seen:
            break
        chain_rev.append(pidx)
        seen.add(pidx)
        cur_parent = parent_id_all[pidx]
        hop += 1
        if hop > 30:
            break
    return list(reversed(chain_rev))

def evaluate_one_solution(y_manage, sO, sI, sD):
    y_manage = np.asarray(y_manage, dtype=int)
    sO = np.asarray(sO, dtype=int)
    sI = np.asarray(sI, dtype=int)
    sD = np.asarray(sD, dtype=int)

    sO = y_manage * sO
    sI = y_manage * sI
    sD = y_manage * sD

    for i in range(m_cand):
        required_additional_floor = int(additional_prot_floor_c[i])
        if y_manage[i] > 0 and required_additional_floor > 0:
            if sO[i] + sI[i] + sD[i] < required_additional_floor:
                if echelon_c[i] == "O":
                    sO[i] = required_additional_floor
                elif echelon_c[i] == "I":
                    sI[i] = required_additional_floor
                else:
                    sD[i] = required_additional_floor

    total_stock = sO + sI + sD

    if APPLY_INITIAL_STOCK_TO_SHORTAGE:
        initial_credit = initial_stock_units_c.copy()
    else:
        initial_credit = np.zeros(m_cand, dtype=int)

    for i in range(m_cand):
        if total_stock[i] > stock_cap_c[i]:
            excess = total_stock[i] - stock_cap_c[i]
            reduc = min(sD[i], excess)
            sD[i] -= reduc
            excess -= reduc
            if excess > 0:
                reduc = min(sI[i], excess)
                sI[i] -= reduc
                excess -= reduc
            if excess > 0:
                reduc = min(sO[i], excess)
                sO[i] -= reduc
                excess -= reduc
        total_stock[i] = sO[i] + sI[i] + sD[i]

    # R-4: Initial_Stock is treated as already-available O-level serviceable stock.
    # It reduces shortage probability but is not counted as newly purchased stock.
    effective_sO = sO + initial_credit
    effective_sI = sI.copy()
    effective_sD = sD.copy()
    effective_total_stock = effective_sO + effective_sI + effective_sD
    has_shortage_credit = (y_manage > 0) | (initial_credit > 0)

    p_short_O = np.zeros(m_cand, dtype=float)
    p_short_I = np.zeros(m_cand, dtype=float)
    p_short_D = np.zeros(m_cand, dtype=float)

    if SHORTAGE_PROB_MODE == "lead_time_demand":
        improved_wait = np.zeros(m_cand, dtype=float)
        for i in range(m_cand):
            if not has_shortage_credit[i]:
                p_short_O[i] = 1.0
                p_short_I[i] = 1.0
                p_short_D[i] = 1.0
                improved_wait[i] = baseline_wait_c[i]
                continue

            mu_lead = lambda_eff_demand_c[i] * shortage_exposure_h_c[i] / HOURS_PER_YEAR

            # Cumulative-stock logic preserves the original O/I/D wait path:
            # O stock covers first; if O is depleted, O+I stock is checked;
            # if O+I is depleted, O+I+D stock is checked.
            p_no_O = stockout_prob_from_lead_demand(mu_lead, effective_sO[i])
            p_no_OI = stockout_prob_from_lead_demand(mu_lead, effective_sO[i] + effective_sI[i])
            p_no_OID = stockout_prob_from_lead_demand(mu_lead, effective_total_stock[i])

            # Guard against numerical monotonicity noise.
            p_no_OI = min(max(p_no_OI, 0.0), p_no_O)
            p_no_OID = min(max(p_no_OID, 0.0), p_no_OI)

            p_short_O[i] = p_no_O
            p_short_I[i] = p_no_OI
            p_short_D[i] = p_no_OID

            prob_O = 1.0 - p_no_O
            prob_I = p_no_O - p_no_OI
            prob_D = p_no_OI - p_no_OID
            prob_baseline = p_no_OID

            improved_wait[i] = (
                prob_O * T_issue_O_c[i]
                + prob_I * T_issue_I_c[i]
                + prob_D * T_issue_D_c[i]
                + prob_baseline * baseline_wait_c[i]
            )

    elif SHORTAGE_PROB_MODE == "legacy_issue_time":
        for i in range(m_cand):
            if not has_shortage_credit[i]:
                p_short_O[i] = 1.0
                p_short_I[i] = 1.0
                p_short_D[i] = 1.0
                continue

            muO = lambda_eff_demand_c[i] * T_issue_O_c[i] / HOURS_PER_YEAR
            muI = lambda_eff_demand_c[i] * T_issue_I_c[i] / HOURS_PER_YEAR
            muD = lambda_eff_demand_c[i] * T_issue_D_c[i] / HOURS_PER_YEAR

            p_short_O[i] = poisson_tail_prob(muO, effective_sO[i])
            p_short_I[i] = poisson_tail_prob(muI, effective_sI[i])
            p_short_D[i] = poisson_tail_prob(muD, effective_sD[i])

        improved_wait = (
            (1.0 - p_short_O) * T_issue_O_c
            + p_short_O * (1.0 - p_short_I) * T_issue_I_c
            + p_short_O * p_short_I * (1.0 - p_short_D) * T_issue_D_c
            + p_short_O * p_short_I * p_short_D * baseline_wait_c
        )
    else:
        raise ValueError("SHORTAGE_PROB_MODE must be 'lead_time_demand' or 'legacy_issue_time'.")

    T_wait_per_failure = np.where(
        has_shortage_credit,
        np.minimum(baseline_wait_c, improved_wait),
        baseline_wait_c
    )

    DT_diag = np.sum(lambda_eff_system_c * T_diag_c * critical_weight_c)
    DT_restore = np.sum(lambda_eff_system_c * T_restore_intrinsic_c * critical_weight_c)
    DT_wait = np.sum(lambda_eff_system_c * T_wait_per_failure * critical_weight_c)
    DT_total = DT_diag + DT_wait + DT_restore
    DT_diag_ratio = safe_ratio(DT_diag, DT_total)
    DT_wait_ratio = safe_ratio(DT_wait, DT_total)
    DT_restore_ratio = safe_ratio(DT_restore, DT_total)

    Ao = float(T_OBS_HOURS / (T_OBS_HOURS + DT_total))

    # R-5 cost policy:
    # - total_stock is the NSGA-added stock selected by the optimizer.
    # - effective_total_stock includes Initial_Stock already available at analysis start.
    # - By default, purchase cost excludes Initial_Stock because it is already owned.
    purchase_stock = total_stock.copy()
    if APPLY_INITIAL_STOCK_TO_COST and CHARGE_INITIAL_STOCK_PURCHASE_COST:
        purchase_cost_basis_stock = effective_total_stock
    else:
        purchase_cost_basis_stock = purchase_stock

    if APPLY_INITIAL_STOCK_TO_COST and HOLD_COST_ON_EFFECTIVE_STOCK:
        hold_cost_basis_stock = effective_total_stock
    else:
        hold_cost_basis_stock = purchase_stock

    stock_cost = float(np.sum(purchase_cost_basis_stock * proc_cost_c))
    hold_cost = float(h_pre * np.sum(hold_cost_basis_stock * proc_cost_c))
    total_cost = stock_cost + hold_cost + BASE_MAINT_SPEND

    managed_items = int(np.sum((y_manage > 0) & (total_stock > 0)))
    total_stock_units = int(np.sum(total_stock))
    purchase_stock_units = int(np.sum(purchase_stock))
    purchase_cost_basis_units = int(np.sum(purchase_cost_basis_stock))
    hold_cost_basis_units = int(np.sum(hold_cost_basis_stock))
    effective_total_stock_units = int(np.sum(effective_total_stock))
    prebuy_selected = int(np.sum((y_manage > 0) & prebuy_flag_c))
    protection_selected = int(np.sum((total_stock > 0) & protection_flag_c))
    protection_effective_selected = int(np.sum((effective_total_stock > 0) & protection_flag_c))
    protection_satisfied_by_initial = int(np.sum((initial_credit >= prot_floor_c) & (prot_floor_c > 0)))
    additional_protection_stock_units = int(np.sum(np.minimum(total_stock, additional_prot_floor_c)))

    return {
        "Ao": Ao,
        "DT_diag_h": float(DT_diag),
        "DT_wait_h": float(DT_wait),
        "DT_restore_h": float(DT_restore),
        "DT_total_h": float(DT_total),
        "DT_diag_ratio": float(DT_diag_ratio),
        "DT_wait_ratio": float(DT_wait_ratio),
        "DT_restore_ratio": float(DT_restore_ratio),
        "T_wait_per_failure": T_wait_per_failure,
        "p_short_O": p_short_O,
        "p_short_I": p_short_I,
        "p_short_D": p_short_D,
        "total_cost": total_cost,
        "stock_cost": stock_cost,
        "hold_cost": hold_cost,
        "sO": sO,
        "sI": sI,
        "sD": sD,
        "total_stock": total_stock,
        "purchase_stock": purchase_stock,
        "purchase_cost_basis_stock": purchase_cost_basis_stock,
        "hold_cost_basis_stock": hold_cost_basis_stock,
        "initial_stock_credit": initial_credit,
        "effective_sO": effective_sO,
        "effective_sI": effective_sI,
        "effective_sD": effective_sD,
        "effective_total_stock": effective_total_stock,
        "managed_items": managed_items,
        "total_stock_units": total_stock_units,
        "purchase_stock_units": purchase_stock_units,
        "purchase_cost_basis_units": purchase_cost_basis_units,
        "hold_cost_basis_units": hold_cost_basis_units,
        "effective_total_stock_units": effective_total_stock_units,
        "prebuy_selected": prebuy_selected,
        "protection_selected": protection_selected,
        "protection_effective_selected": protection_effective_selected,
        "protection_satisfied_by_initial": protection_satisfied_by_initial,
        "additional_protection_stock_units": additional_protection_stock_units,
    }

# 10) NSGA evaluation + utilities
# ============================================================
def eval_nsga(pop_mat, ao_target):
    pop = pop_mat.shape[0]

    Y = np.clip(np.rint(pop_mat[:, :m_cand]), 0, 1).astype(int)
    SO = np.clip(np.rint(pop_mat[:, m_cand:m_cand+m_cand]), 0, None).astype(int)
    SI = np.clip(np.rint(pop_mat[:, m_cand+m_cand:m_cand+2*m_cand]), 0, None).astype(int)
    SD = np.clip(np.rint(pop_mat[:, m_cand+2*m_cand:m_cand+3*m_cand]), 0, None).astype(int)

    F = np.zeros((pop, 3), dtype=float)
    Ao = np.zeros(pop, dtype=float)

    for p in range(pop):
        sim = evaluate_one_solution(Y[p], SO[p], SI[p], SD[p])

        Ao[p] = sim["Ao"]
        ao_shortfall = max(0.0, ao_target - sim["Ao"])

        F1 = sim["total_cost"] + AO_COST_PENALTY * (ao_shortfall ** 2)
        F2 = ao_shortfall
        F3 = ITEM_COUNT_WEIGHT * sim["managed_items"] + STOCK_UNIT_WEIGHT * sim["total_stock_units"]

        F[p, 0] = F1
        F[p, 1] = F2
        F[p, 2] = F3

    return F, Ao

def dominates(a, b):
    return np.all(a <= b) and np.any(a < b)

def fast_non_dominated_sort(F):
    n = F.shape[0]
    S = [[] for _ in range(n)]
    n_dom = np.zeros(n, dtype=int)
    fronts = [[]]

    for p in range(n):
        for q in range(n):
            if p == q:
                continue
            if dominates(F[p], F[q]):
                S[p].append(q)
            elif dominates(F[q], F[p]):
                n_dom[p] += 1
        if n_dom[p] == 0:
            fronts[0].append(p)

    i = 0
    while fronts[i]:
        nxt = []
        for p in fronts[i]:
            for q in S[p]:
                n_dom[q] -= 1
                if n_dom[q] == 0:
                    nxt.append(q)
        i += 1
        fronts.append(nxt)

    fronts.pop()
    return fronts

def crowding_distance(F, front):
    mobj = F.shape[1]
    dist = np.zeros(len(front), dtype=float)
    if len(front) == 0:
        return dist

    idx = np.array(front, dtype=int)
    for j in range(mobj):
        vals = F[idx, j]
        order = np.argsort(vals)
        dist[order[0]] = np.inf
        dist[order[-1]] = np.inf

        vmin, vmax = vals[order[0]], vals[order[-1]]
        if vmax - vmin < 1e-12:
            continue

        for k in range(1, len(front) - 1):
            dist[order[k]] += (vals[order[k+1]] - vals[order[k-1]]) / (vmax - vmin)
    return dist

def rank_and_crowd(F):
    fronts = fast_non_dominated_sort(F)
    rank = np.empty(F.shape[0], dtype=int)
    crowd = np.zeros(F.shape[0], dtype=float)

    for rnk, front in enumerate(fronts):
        for i in front:
            rank[i] = rnk
        cd = crowding_distance(F, front)
        for t, i in enumerate(front):
            crowd[i] = cd[t]
    return rank, crowd, fronts

def tournament_select(rank, crowd, rng):
    i, j = rng.integers(0, len(rank), size=2)
    if rank[i] < rank[j]:
        return i
    if rank[j] < rank[i]:
        return j
    return i if crowd[i] > crowd[j] else j

def uniform_crossover(p1, p2, rng):
    mask = rng.random(p1.shape[0]) < 0.5
    c1, c2 = p1.copy(), p2.copy()
    c1[mask] = p2[mask]
    c2[mask] = p1[mask]
    return c1, c2

# 11) Run NSGA-II (single target)
# ============================================================
def run_nsga_single_target(ao_target, seed=42):
    rng = np.random.default_rng(seed)

    n_vars = m_cand + 3 * m_cand
    mut_p = float(np.clip(20.0 / max(n_vars, 1), 0.01, 0.06))

    pop = np.zeros((POP, n_vars), dtype=float)

    seed_prob = np.clip(
        0.12 + 0.55 * (
            0.45 * minmax(p_need_c) +
            0.35 * minmax(impact_c) +
            0.20 * is_long_c.astype(float)
        ),
        0.08, 0.85
    )

    for k in range(POP):
        for j in range(m_cand):
            pop[k, j] = 1 if rng.random() < seed_prob[j] else 0

        for j in range(m_cand):
            yj = int(pop[k, j])
            ub = stock_cap_c[j]
            if yj == 0:
                pop[k, m_cand + j] = 0
                pop[k, m_cand + m_cand + j] = 0
                pop[k, m_cand + 2*m_cand + j] = 0
            else:
                sO0, sI0, sD0 = 0, 0, 0
                if echelon_c[j] == "O":
                    sO0 = rng.integers(additional_prot_floor_c[j], min(ub, max(1, additional_prot_floor_c[j])) + 1)
                elif echelon_c[j] == "I":
                    sI0 = rng.integers(additional_prot_floor_c[j], min(ub, max(1, additional_prot_floor_c[j])) + 1)
                else:
                    sD0 = rng.integers(additional_prot_floor_c[j], min(ub, max(1, additional_prot_floor_c[j])) + 1)
                pop[k, m_cand + j] = sO0
                pop[k, m_cand + m_cand + j] = sI0
                pop[k, m_cand + 2*m_cand + j] = sD0

    def repair_solution(v):
        y = np.clip(np.rint(v[:m_cand]), 0, 1).astype(int)
        sO = np.clip(np.rint(v[m_cand:m_cand+m_cand]), 0, None).astype(int)
        sI = np.clip(np.rint(v[m_cand+m_cand:m_cand+2*m_cand]), 0, None).astype(int)
        sD = np.clip(np.rint(v[m_cand+2*m_cand:m_cand+3*m_cand]), 0, None).astype(int)

        for j in range(m_cand):
            if y[j] == 0:
                sO[j] = 0
                sI[j] = 0
                sD[j] = 0
            total = sO[j] + sI[j] + sD[j]
            cap = stock_cap_c[j]
            if total > cap:
                excess = total - cap
                reduc = min(sD[j], excess)
                sD[j] -= reduc
                excess -= reduc
                if excess > 0:
                    reduc = min(sI[j], excess)
                    sI[j] -= reduc
                    excess -= reduc
                if excess > 0:
                    reduc = min(sO[j], excess)
                    sO[j] -= reduc
                    excess -= reduc

            add_floor = int(additional_prot_floor_c[j])
            if y[j] > 0 and add_floor > 0 and (sO[j] + sI[j] + sD[j]) < add_floor:
                add_need = min(add_floor, cap)
                if echelon_c[j] == "O":
                    sO[j] = max(sO[j], add_need)
                elif echelon_c[j] == "I":
                    sI[j] = max(sI[j], add_need)
                else:
                    sD[j] = max(sD[j], add_need)

        return np.concatenate([y, sO, sI, sD]).astype(float)

    def mutate(child):
        y = child.copy()

        msk_y = rng.random(m_cand) < mut_p
        for j in range(m_cand):
            if msk_y[j]:
                y[j] = 1.0 - round(y[j])

        for block in range(3):
            start = m_cand + block * m_cand
            msk_s = rng.random(m_cand) < mut_p
            for j in range(m_cand):
                idx = start + j
                if msk_s[j]:
                    ub = stock_cap_c[j]
                    y[idx] = rng.integers(0, ub + 1)

        y = repair_solution(y)
        return y

    for gen in range(GEN):
        F, Ao = eval_nsga(pop, ao_target)
        rank, crowd, fronts = rank_and_crowd(F)

        off = []
        while len(off) < POP:
            p1 = tournament_select(rank, crowd, rng)
            p2 = tournament_select(rank, crowd, rng)
            c1, c2 = uniform_crossover(pop[p1], pop[p2], rng)
            c1 = mutate(c1)
            c2 = mutate(c2)
            off.append(c1)
            if len(off) < POP:
                off.append(c2)
        off = np.array(off, dtype=float)

        comb = np.vstack([pop, off])
        F_c, Ao_c = eval_nsga(comb, ao_target)
        rank_c, crowd_c, fronts_c = rank_and_crowd(F_c)

        new_idx = []
        for front in fronts_c:
            if len(new_idx) + len(front) <= POP:
                new_idx.extend(front)
            else:
                front = np.array(front, dtype=int)
                order = np.argsort(-crowd_c[front])
                need = POP - len(new_idx)
                new_idx.extend(front[order[:need]].tolist())
                break

        pop = comb[np.array(new_idx, dtype=int)]

        if (gen + 1) % 10 == 0:
            Fp, Aop = eval_nsga(pop, ao_target)
            managed_counts = []
            for row in pop:
                ytmp = np.clip(np.rint(row[:m_cand]), 0, 1).astype(int)
                managed_counts.append(int(np.sum(ytmp)))
            print(
                f"[NSGA | target={ao_target:.2f}] Gen {gen+1}/{GEN} | "
                f"best Ao={float(np.max(Aop)):.4f} | "
                f"share(Ao>={ao_target:.2f})={float(np.mean(Aop >= ao_target)):.2f} | "
                f"mean managed={float(np.mean(managed_counts)):.1f}"
            )

    F, Ao = eval_nsga(pop, ao_target)
    cost = F[:, 0]

    F2d = np.column_stack([cost, -Ao])
    fronts2d = fast_non_dominated_sort(F2d)
    pareto_idx = np.array(fronts2d[0], dtype=int)
    dominated_idx = np.setdiff1d(np.arange(len(cost)), pareto_idx)

    order = np.argsort(cost[pareto_idx])
    p_idx = pareto_idx[order]

    best_idx = int(np.argmax(Ao))
    x = pop[best_idx]

    Y = np.clip(np.rint(x[:m_cand]), 0, 1).astype(int)
    SO = np.clip(np.rint(x[m_cand:m_cand+m_cand]), 0, None).astype(int)
    SI = np.clip(np.rint(x[m_cand+m_cand:m_cand+2*m_cand]), 0, None).astype(int)
    SD = np.clip(np.rint(x[m_cand+2*m_cand:m_cand+3*m_cand]), 0, None).astype(int)

    best_sim = evaluate_one_solution(Y, SO, SI, SD)

    return {
        "target_ao": ao_target,
        "pop": pop,
        "F": F,
        "Ao": Ao,
        "cost": cost,
        "pareto_idx": pareto_idx,
        "dominated_idx": dominated_idx,
        "p_cost": cost[p_idx],
        "p_Ao": Ao[p_idx],
        "best_Ao": float(np.max(Ao)),
        "share_ge_target": float(np.mean(Ao >= ao_target)),
        "best_sim": best_sim,
        "best_y": Y,
        "best_sO": SO,
        "best_sI": SI,
        "best_sD": SD,
    }

def run_target_sweep(target_grid, base_seed=42):
    all_result_map = {}
    all_df = []

    for idx, target in enumerate(target_grid):
        print("\n" + "=" * 70)
        print(f"[TARGET SWEEP] RUN START | target_ao = {target:.2f}")
        print("=" * 70)

        run_seed = int(base_seed + idx * 17)

        res_t = run_nsga_single_target(ao_target=target, seed=run_seed)
        all_result_map[round(target, 6)] = res_t

        df_t = collect_run_dataframe(res_t)
        all_df.append(df_t)

        print(f"[TARGET SUMMARY] target={target:.2f} | best Ao={res_t['best_Ao']:.4f} | share_ge_target={res_t['share_ge_target']:.2f}")

    if len(all_df) == 0:
        return all_result_map, pd.DataFrame()

    df_all = pd.concat(all_df, ignore_index=True)
    return all_result_map, df_all

def deduplicate_integrated_df(df_all):
    if df_all.empty:
        return df_all.copy()

    df_unique = (
        df_all.sort_values(
            by=["Ao", "total_cost", "target_ao"],
            ascending=[False, True, True]
        )
        .drop_duplicates(subset=["signature"])
        .reset_index(drop=True)
    )
    return df_unique

def plot_representative_pareto(res_single):
    plt.figure(figsize=(9, 6))

    plt.scatter(
        res_single["cost"][res_single["dominated_idx"]],
        res_single["Ao"][res_single["dominated_idx"]],
        s=18,
        alpha=0.35,
        color="gray",
        label="Dominated solutions"
    )

    plt.scatter(
        res_single["p_cost"],
        res_single["p_Ao"],
        s=34,
        alpha=0.95,
        color="#1E88E5",
        label="Pareto solutions"
    )
    plt.plot(
        res_single["p_cost"],
        res_single["p_Ao"],
        linewidth=2.2,
        color="#1E88E5",
        label="Pareto front"
    )

    for ao_line in AO_LINES:
        color = "#2E7D32" if ao_line < res_single["target_ao"] else "#6A1B9A"
        style = "--" if ao_line < res_single["target_ao"] else "-."
        plt.axhline(
            ao_line,
            linestyle=style,
            linewidth=1.1,
            color=color,
            alpha=0.85
        )
        plt.text(
            x=np.nanmin(res_single["cost"]) * 1.001,
            y=ao_line + 0.0008,
            s=f"Ao={ao_line:.2f}",
            fontsize=8,
            color=color
        )

    plt.xlabel("Total Cost (KRW)")
    plt.ylabel("Ao")
    plt.title(f"Representative Pareto-front (target={res_single['target_ao']:.2f})")
    plt.legend(loc="best")
    plt.tight_layout()
    plt.show()

def plot_targetwise_pareto_overlay(result_map):
    plt.figure(figsize=(10, 6))

    cmap = plt.cm.viridis
    targets_sorted = sorted(result_map.keys())

    for idx, t in enumerate(targets_sorted):
        res_t = result_map[t]
        color = cmap(idx / max(len(targets_sorted) - 1, 1))

        plt.scatter(
            res_t["p_Ao"],
            res_t["p_cost"],
            s=18,
            alpha=0.75,
            color=color,
            label=f"target={t:.2f}"
        )
        plt.plot(
            res_t["p_Ao"],
            res_t["p_cost"],
            linewidth=1.1,
            alpha=0.75,
            color=color
        )

    for ao_line in AO_LINES:
        plt.axvline(
            ao_line,
            linestyle="--",
            linewidth=0.9,
            alpha=0.5
        )

    plt.xlabel("Ao")
    plt.ylabel("Total Cost (KRW)")
    plt.title("Overlay of Pareto fronts by target")
    plt.grid(True, alpha=0.25)
    plt.legend(loc="best", fontsize=8)
    plt.tight_layout()
    plt.show()
