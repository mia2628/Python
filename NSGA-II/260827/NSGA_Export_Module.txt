def _require_globals(ns: Dict[str, Any], names: Sequence[str]) -> None:
    missing = [name for name in names if name not in ns]
    if missing:
        raise RuntimeError(
            "Excel report export requires the NSGA-II model to be executed first "
            "in the same Jupyter kernel. Missing variables: " + ", ".join(missing)
        )


def _safe_float(value: Any, default: float = 0.0) -> float:
    try:
        v = float(value)
        return v if np.isfinite(v) else default
    except Exception:
        return default


def _safe_int(value: Any, default: int = 0) -> int:
    try:
        return int(round(float(value)))
    except Exception:
        return default


def _safe_div(a: float, b: float) -> float:
    return float(a / b) if b not in (0, 0.0) and np.isfinite(b) else 0.0


def _excel_col_width(series: pd.Series, header: str, min_w: int = 10, max_w: int = 34) -> int:
    if len(series) == 0:
        return max(min_w, min(max_w, len(str(header)) + 2))
    sample = series.astype(str).replace("nan", "").head(1000)
    max_len = max([len(str(header))] + [len(x) for x in sample.tolist()])
    return max(min_w, min(max_w, max_len + 2))

def _write_title(ws: xlsxwriter.worksheet.Worksheet, fmt: Dict[str, Any], title: str, subtitle: str = "") -> None:
    ws.set_row(0, 34)
    ws.merge_range("A1:N1", title, fmt["title"])
    ws.set_row(1, 20)
    ws.merge_range("A2:N2", subtitle, fmt["subtitle"])
    ws.hide_gridlines(2)


def _write_section(ws: xlsxwriter.worksheet.Worksheet, fmt: Dict[str, Any], row: int, title: str, last_col: int = 13) -> None:
    ws.set_row(row, 22)
    ws.merge_range(row, 0, row, last_col, title, fmt["section"])


def _write_kpi(
    ws: xlsxwriter.worksheet.Worksheet,
    fmt: Dict[str, Any],
    top_row: int,
    left_col: int,
    label: str,
    value: Any,
    kind: str = "number",
    span: int = 2,
) -> None:
    right_col = left_col + span - 1
    ws.merge_range(top_row, left_col, top_row, right_col, label, fmt["kpi_label"])
    f = {
        "percent": fmt["kpi_percent"],
        "money": fmt["kpi_money"],
        "number": fmt["kpi_number"],
        "text": fmt["kpi_value"],
    }.get(kind, fmt["kpi_value"])
    ws.merge_range(top_row + 1, left_col, top_row + 2, right_col, value, f)
    ws.set_row(top_row, 18)
    ws.set_row(top_row + 1, 23)
    ws.set_row(top_row + 2, 23)


def _format_dataframe_columns(
    ws: xlsxwriter.worksheet.Worksheet,
    df: pd.DataFrame,
    start_row: int,
    start_col: int,
    formats: Dict[str, Any],
) -> None:
    for j, col in enumerate(df.columns):
        col_l = str(col).lower()
        series = df[col]
        width = _excel_col_width(series, str(col))
        cell_format = formats["text"]

        if any(k in col_l for k in ["cost", "price", "spend", "krw"]):
            cell_format = formats["money"]
            width = max(width, 15)
        elif col_l == "ao" or col_l.startswith("ao_") and any(k in col_l for k in ["share", "ratio"]):
            cell_format = formats["percent"]
            width = max(width, 12)
        elif any(k in col_l for k in ["ratio", "share_ge", "p_need", "p_short"]):
            cell_format = formats["percent"]
            width = max(width, 12)
        elif any(k in col_l for k in ["_h", "hours", "wait", "restore", "diag", "t_obs"]):
            if pd.api.types.is_numeric_dtype(series):
                cell_format = formats["hours"]
                width = max(width, 13)
        elif pd.api.types.is_integer_dtype(series):
            cell_format = formats["int"]
        elif pd.api.types.is_numeric_dtype(series):
            cell_format = formats["decimal"]
        elif any(k in col_l for k in ["reason", "source"]):
            cell_format = formats["text_wrap"]
            width = max(width, 22)

        ws.set_column(start_col + j, start_col + j, width, cell_format)


def _write_df_table(
    workbook: xlsxwriter.Workbook,
    ws: xlsxwriter.worksheet.Worksheet,
    df: pd.DataFrame,
    start_row: int,
    start_col: int,
    table_name: str,
    formats: Dict[str, Any],
    max_rows: Optional[int] = None,
) -> Tuple[int, int]:
    work_df = df.copy()
    if max_rows is not None:
        work_df = work_df.head(max_rows).copy()

    # Normalize NaN/Inf for Excel stability.
    work_df = work_df.replace([np.inf, -np.inf], np.nan)

    nrows, ncols = work_df.shape
    if ncols == 0:
        return start_row, start_col

    # Header
    for j, col in enumerate(work_df.columns):
        ws.write(start_row, start_col + j, str(col), formats["header"])

    # Data
    for i, row in enumerate(work_df.itertuples(index=False, name=None), start=1):
        for j, val in enumerate(row):
            if pd.isna(val):
                ws.write_blank(start_row + i, start_col + j, None)
            elif isinstance(val, (np.integer, int)) and not isinstance(val, bool):
                ws.write_number(start_row + i, start_col + j, int(val))
            elif isinstance(val, (np.floating, float)):
                ws.write_number(start_row + i, start_col + j, float(val))
            else:
                ws.write(start_row + i, start_col + j, str(val))

    if nrows > 0:
        ws.add_table(
            start_row,
            start_col,
            start_row + nrows,
            start_col + ncols - 1,
            {
                "name": table_name,
                "style": "Table Style Medium 2",
                "columns": [{"header": str(c)} for c in work_df.columns],
            },
        )
    _format_dataframe_columns(ws, work_df, start_row, start_col, formats)
    ws.set_row(start_row, 34)
    return start_row + nrows, start_col + ncols - 1

def _pick_representative_best_row(df_all_runs: pd.DataFrame, target: float) -> pd.Series:
    g = df_all_runs[np.isclose(df_all_runs["target_ao"].astype(float), float(target))].copy()
    if g.empty:
        g = df_all_runs.copy()
    feasible = g[g["Ao"].astype(float) >= float(target)].copy()
    if not feasible.empty:
        return feasible.sort_values(["total_cost", "Ao"], ascending=[True, False]).iloc[0]
    return g.sort_values(["Ao", "total_cost"], ascending=[False, True]).iloc[0]

# ======================================================================
def export_nsga_excel_report(
    output_path: Optional[str] = None,
    namespace: Optional[Dict[str, Any]] = None,
) -> str:
    """Create a polished XLSX report from the executed NSGA-II model results."""

    ns = namespace if namespace is not None else globals()
    required = ["summary_by_target", "df_all_runs", "df_integrated", "diag_df", "res", "b"]
    _require_globals(ns, required)

    summary_by_target = ns["summary_by_target"].copy()
    df_all_runs = ns["df_all_runs"].copy()
    df_integrated = ns["df_integrated"].copy()
    diag_df = ns["diag_df"].copy()
    res = ns["res"]
    b = ns["b"]

    years = _safe_float(ns.get("YEARS", b.get("analysis_years", 0.0)))
    t_obs_h = _safe_float(ns.get("T_OBS_HOURS", b.get("T_obs_h", 0.0)))
    rep_target = _safe_float(ns.get("REPRESENTATIVE_TARGET", res.get("target_ao", 0.0)))

    output_path = output_path or REPORT_PATH
    output_dir = os.path.dirname(output_path) or "."
    os.makedirs(output_dir, exist_ok=True)

    # ------------------------------------------------------------------
    # Derived report datasets
    # ------------------------------------------------------------------
    best_ao = _safe_float(b.get("Ao", res.get("best_Ao", 0.0)))
    target_met = best_ao >= rep_target

    # Final-policy quantities include the policy overlay, while pure NSGA values remain separate.
    final_initial_rows = _safe_int((diag_df.get("final_initial_spare_qty", pd.Series(dtype=float)) > 0).sum()) if "final_initial_spare_qty" in diag_df else 0
    final_initial_units = _safe_int(diag_df.get("final_initial_spare_qty", pd.Series(dtype=float)).sum()) if "final_initial_spare_qty" in diag_df else 0
    policy_added_units = _safe_int(diag_df.get("policy_added_qty", pd.Series(dtype=float)).sum()) if "policy_added_qty" in diag_df else 0
    policy_added_rows = _safe_int((diag_df.get("policy_added_qty", pd.Series(dtype=float)) > 0).sum()) if "policy_added_qty" in diag_df else 0

    stock_cost = _safe_float(b.get("stock_cost", 0.0))
    hold_cost = _safe_float(b.get("hold_cost", 0.0))
    total_cost = _safe_float(b.get("total_cost", 0.0))
    base_maint_spend = _safe_float(ns.get("BASE_MAINT_SPEND", max(total_cost - stock_cost - hold_cost, 0.0)))

    dt_diag_annual = _safe_float(b.get("DT_diag_annual_h", 0.0))
    dt_wait_annual = _safe_float(b.get("DT_wait_annual_h", 0.0))
    dt_restore_annual = _safe_float(b.get("DT_restore_annual_h", 0.0))
    dt_total_annual = _safe_float(b.get("DT_total_annual_h", dt_diag_annual + dt_wait_annual + dt_restore_annual))
    dt_diag = _safe_float(b.get("DT_diag_h", dt_diag_annual * years))
    dt_wait = _safe_float(b.get("DT_wait_h", dt_wait_annual * years))
    dt_restore = _safe_float(b.get("DT_restore_h", dt_restore_annual * years))
    dt_total = _safe_float(b.get("DT_total_h", dt_diag + dt_wait + dt_restore))

    dt_analysis = pd.DataFrame([
        {"DT_Component": "Diagnosis", "Annual_DT_h": dt_diag_annual, "Period_DT_h": dt_diag, "Share": _safe_div(dt_diag, dt_total)},
        {"DT_Component": "Stock/Logistics Wait", "Annual_DT_h": dt_wait_annual, "Period_DT_h": dt_wait, "Share": _safe_div(dt_wait, dt_total)},
        {"DT_Component": "Restore/Repair", "Annual_DT_h": dt_restore_annual, "Period_DT_h": dt_restore, "Share": _safe_div(dt_restore, dt_total)},
        {"DT_Component": "TOTAL", "Annual_DT_h": dt_total_annual, "Period_DT_h": dt_total, "Share": 1.0 if dt_total > 0 else 0.0},
    ])

    cost_analysis = pd.DataFrame([
        {"Cost_Component": "Initial Stock Cost", "Amount_KRW": stock_cost, "Share": _safe_div(stock_cost, total_cost)},
        {"Cost_Component": "Holding Cost", "Amount_KRW": hold_cost, "Share": _safe_div(hold_cost, total_cost)},
        {"Cost_Component": "Expected Maintenance Spend", "Amount_KRW": base_maint_spend, "Share": _safe_div(base_maint_spend, total_cost)},
        {"Cost_Component": "TOTAL", "Amount_KRW": total_cost, "Share": 1.0 if total_cost > 0 else 0.0},
    ])

    # Best feasible/near-feasible solution by target for a compact decision table.
    decision_rows = []

    # ==================================================================
    ws = workbook.add_worksheet(SHEET_SETTINGS)
    ws.set_tab_color(C["navy"])
    _write_title(ws, fmt, "모델 설정 및 추적정보", "재현성을 위해 분석기간, 고장률 단위, NSGA 파라미터, 정책 임계값과 입력파일 정보를 기록합니다.")

    _write_section(ws, fmt, 3, "Model / Run Settings", 6)
    _write_df_table(workbook, ws, settings_df, 5, 0, "ModelSettingsTable", fmt)
    ws.set_column("A:A", 38)
    ws.set_column("B:B", 80, fmt["text_wrap"])

    # ==================================================================
    # Common finishing touches
    # ==================================================================
    # Ensure every sheet starts with a clean zoom and useful print behavior.
    for sname in [
        SHEET_EXECUTIVE, SHEET_TARGET, SHEET_PARETO, SHEET_POLICY,
        SHEET_DIAG, SHEET_DT, SHEET_COST, SHEET_SETTINGS,
    ]:
        sheet = workbook.get_worksheet_by_name(sname)
        sheet.set_zoom(90)
        sheet.set_landscape()
        sheet.fit_to_pages(1, 0)
        sheet.set_margins(left=0.25, right=0.25, top=0.5, bottom=0.5)
        sheet.set_header("&L" + REPORT_TITLE + "&R" + sname)
        sheet.set_footer("&LGenerated by NSGA-II Report Export&RPage &P of &N")

    workbook.close()

    if not os.path.exists(output_path) or os.path.getsize(output_path) <= 0:
        raise IOError(f"Excel report was not created correctly: {output_path}")

    print("Sheets :")
    for s in [
        SHEET_EXECUTIVE, SHEET_TARGET, SHEET_PARETO, SHEET_POLICY,
        SHEET_DIAG, SHEET_DT, SHEET_COST, SHEET_SETTINGS,
    ]:
        print(f"  - {s}")
    print("=" * 72)
    return output_path
